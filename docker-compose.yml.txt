services:
  mysql:
    image: mysql:8.0
    container_name: espo-mysql
    environment:
      MYSQL_ROOT_PASSWORD: your_strong_password
      MYSQL_DATABASE: espocrm
    volumes:
      - mysql_data:/var/lib/mysql
    restart: always

  espocrm:
    image: espocrm/espocrm
    container_name: espo-crm
    environment:
      ESPOCRM_DATABASE_HOST: mysql
      ESPOCRM_DATABASE_USER: root
      ESPOCRM_DATABASE_PASSWORD: your_strong_password
      APACHE_MPM: prefork
    ports:
      - "8080:80"
    volumes:
      - espo_data:/var/www/html
    restart: always
    depends_on:
      - mysql

volumes:
  mysql_data:
